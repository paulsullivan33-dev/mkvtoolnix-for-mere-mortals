# Recipe A: add a second subtitle track to the finished film.
# The existing MKV is just another input file to mkvmerge.
$ mkvmerge -o work/with-largeprint.mkv work/finished.mkv \
    --track-name 0:"Subtitles (large print)" --language 0:eng \
    fixtures/sample-subs-largeprint.srt
exit: 0
$ python3 track-table.py work/with-largeprint.mkv
